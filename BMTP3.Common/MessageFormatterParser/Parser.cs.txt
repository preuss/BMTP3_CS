using BMTP3.Common.MessageFormatterParser.Nodes;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace BMTP3.Common.MessageFormatterParser {
	public class Parser {
		private readonly List<Token> tokens;
		private int position;

		public Parser(List<Token> tokens) {
			this.tokens = tokens;
			position = 0;
		}

		public RootNode Parse() {
			var root = new RootNode();
			while(position < tokens.Count) {
				var token = tokens[position];
				if(token.Type == TokenType.Text) {
					root.Children.Add(new TextNode(token.Value));
					position++;
				} else if(token.Type == TokenType.PlaceholderStart) {
					position++; // Skip ${ or #{
					var placeholder = ParsePlaceholder();
					root.Children.Add(placeholder);
				} else {
					throw new Exception($"Unexpected token {token} at position {token.Position}");
				}
			}
			return root;
		}

		private PlaceholderNode ParsePlaceholder() {
			if(position >= tokens.Count || tokens[position].Type != TokenType.Identifier) {
				throw new Exception($"Expected identifier at position {(position < tokens.Count ? tokens[position].Position : "EOF")}");
			}

			var nameOrIndex = tokens[position].Value;
			position++;

			var functions = new List<FunctionCallNode>();
			string? type = null;
			string? style = null;
			List<AstNode>? pattern = null;
			IfConditionNode? condition = null;

			// Parse function calls
			while(position < tokens.Count && tokens[position].Type == TokenType.Dot) {
				position++; // Skip .
				if(position >= tokens.Count ||
					tokens[position].Type != TokenType.Identifier
				) {
					throw new Exception($"Expected function name at position {(position < tokens.Count ? tokens[position].Position : "EOF")}");
				}
				var funcName = tokens[position].Value;
				position++;
				string? parameters = null;
				if(position < tokens.Count && tokens[position].Type == TokenType.ParenOpen) {
					position++; // Skip (
					if(position < tokens.Count && tokens[position].Type == TokenType.Parameters) {
						parameters = tokens[position].Value;
						position++;
					}
					if(position >= tokens.Count || tokens[position].Type != TokenType.ParenClose) {
						throw new Exception($"Expected ) at position {(position < tokens.Count ? tokens[position].Position : "EOF")}");
					}
					position++; // Skip )
				}
				functions.Add(new FunctionCallNode(funcName, parameters));
			}

			// Parse type, style, or pattern
			if(position < tokens.Count && tokens[position].Type == TokenType.Comma) {
				position++; // Skip ,
				if(position >= tokens.Count || tokens[position].Type != TokenType.Identifier) {
					throw new Exception($"Expected type at position {(position < tokens.Count ? tokens[position].Position : "EOF")}");
				}
				type = tokens[position].Value;
				position++;

				if(position < tokens.Count && tokens[position].Type == TokenType.Comma) {
					position++; // Skip ,
					if(position >= tokens.Count || tokens[position].Type != TokenType.Identifier) {
						throw new Exception($"Expected style at position {(position < tokens.Count ? tokens[position].Position : "EOF")}");
					}
					style = tokens[position].Value;
					position++;
				}

				if(position < tokens.Count && tokens[position].Type == TokenType.Colon) {
					position++; // Skip :
					pattern = ParsePattern();
				}
			} else if(position < tokens.Count && tokens[position].Type == TokenType.Colon) {
				position++; // Skip :
				pattern = ParsePattern();
			} else if(position < tokens.Count && tokens[position].Type == TokenType.Identifier && tokens[position].Value == "if") {
				condition = ParseIfCondition();
			}

			if(position >= tokens.Count || tokens[position].Type != TokenType.PlaceholderEnd) {
				var positionMessage = position < tokens.Count ? tokens[position].Position.ToString() : "EOF";
				throw new Exception($"Expected '}}' at position {positionMessage}");
			}

			position++; // Skip }

			return new PlaceholderNode(nameOrIndex, type, style, pattern, condition);
		}

		private List<AstNode> ParsePattern() {
			var content = new List<AstNode>();
			while(position < tokens.Count && tokens[position].Type != TokenType.PlaceholderEnd) {
				var token = tokens[position];
				if(token.Type == TokenType.Text || token.Type == TokenType.Identifier || token.Type == TokenType.Parameters || token.Type == TokenType.Comma || token.Type == TokenType.Colon) {
					content.Add(new TextNode(token.Value));
					position++;
				} else if(token.Type == TokenType.EscapedBrace) {
					content.Add(new LiteralNode(token.Value == "{{" ? "{" : "}"));
					position++;
				} else if(token.Type == TokenType.PlaceholderStart) {
					position++; // Skip ${ or #{
					content.Add(ParsePlaceholder());
				} else {
					throw new Exception($"Unexpected token in pattern: {token}");
				}
			}
			return content;
		}

		private IfConditionNode ParseIfCondition() {
			position++; // Skip if
			if(position >= tokens.Count || tokens[position].Type != TokenType.Comma) {
				throw new Exception($"Expected , after if at position {tokens[position].Position}");
			}
			position++; // Skip ,

			if(position >= tokens.Count || tokens[position].Type != TokenType.Identifier) {
				throw new Exception($"Expected condition at position {tokens[position].Position}");
			}
			var condition = tokens[position].Value;
			string? conditionParameters = null;
			position++;

			if(position < tokens.Count && tokens[position].Type == TokenType.ParenOpen) {
				position++; // Skip (
				if(position < tokens.Count && tokens[position].Type == TokenType.Parameters) {
					conditionParameters = tokens[position].Value;
					position++;
				}
				if(position >= tokens.Count || tokens[position].Type != TokenType.ParenClose) {
					throw new Exception($"Expected ) at position {tokens[position].Position}");
				}
				position++; // Skip )
			}

			if(position >= tokens.Count || tokens[position].Type != TokenType.QuestionMark) {
				throw new Exception($"Expected ? at position {tokens[position].Position}");
			}
			position++; // Skip ?

			var trueString = ParsePattern(); // Parse until :

			if(position >= tokens.Count || tokens[position].Type != TokenType.Colon) {
				throw new Exception($"Expected : at position {tokens[position].Position}");
			}
			position++; // Skip :

			var falseString = ParsePattern(); // Parse until }

			return new IfConditionNode(condition, conditionParameters, trueString, falseString);
		}
	}
}
