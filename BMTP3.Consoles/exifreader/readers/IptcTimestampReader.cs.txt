using BMTP3.Consoles.candidates;
using BMTP3.Consoles.exifreader.definitions;
using BMTP3.Consoles.exifreader.parsers;
using MetadataExtractor;
using MetadataExtractor.Formats.Iptc;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace BMTP3.Consoles.exifreader.readers;

foreach (TimestampTagGroup group in TagGroups.IptcTagDefinitions)
{
    // Read raw values
    string? rawDate = GetStringForTag(iptcDir, group.DateTag);
    string? rawTime = GetStringForTag(iptcDir, group.TimeTag);
    string? rawSub = GetStringForTag(iptcDir, group.SubSecTag);
    string? rawOffset = GetStringForTag(iptcDir, group.OffsetTag);
    string? rawTimestamp = GetStringForTag(iptcDir, group.TimestampTag);
    long? timestampValue = GetLongForTag(iptcDir, group.TimestampTag);

    // 1. Priority: Numeric timestamp
    if (timestampValue.HasValue)
    {
        NumericTimestampType numericTimestampType = GuessUnixType(timestampValue.Value, group.TimestampType);
        candidates.Add(TimestampCandidateFactory.FromUtcNumeric(
            TimestampSourceType.Iptc,
            group.Role,
            rawTimestamp,
            timestampValue.Value,
            numericTimestampType
        ));
        continue; // Skip text parsing for this group if we have a numeric timestamp
    }

    // 2. Priority: Text-based cascade parsing on rawDate
    DateOnly? dateResult = null;
    TimeOnly? timeResult = null;
    TimeSpan? offsetResult = null;

    if (!string.IsNullOrWhiteSpace(rawDate))
    {
        if (_dateTimeOffsetParser.TryParse(rawDate, out DateTimeOffset dto))
        {
            dateResult = DateOnly.FromDateTime(dto.DateTime);
            timeResult = new TimeOnly(dto.Hour, dto.Minute, dto.Second);
            offsetResult = dto.Offset;
        }
        else if (_dateTimeParser.TryParse(rawDate, out DateTime dt))
        {
            dateResult = DateOnly.FromDateTime(dt);
            timeResult = new TimeOnly(dt.Hour, dt.Minute, dt.Second);
        }
        else if (_dateParser.TryParse(rawDate, out DateOnly d))
        {
            dateResult = d;
        }
    }

    // 3. Supplemental: If time was not found in rawDate, try rawTime
    if (timeResult == null && !string.IsNullOrWhiteSpace(rawTime))
    {
        if (_timeParser.TryParse(rawTime, out TimeOnly t))
        {
            timeResult = t;
        }
    }

    // 4. Final: Only add candidate if we have at least some raw evidence or a parsed result
    if (!string.IsNullOrWhiteSpace(rawDate) || !string.IsNullOrWhiteSpace(rawTime))
    {
        candidates.Add(TimestampCandidateFactory.FromParts(
            TimestampSourceType.Iptc,
            group.Role,
            new TimestampSources 
            { 
                Date = rawDate, 
                Time = rawTime, 
                SubSeconds = rawSub, 
                Offset = rawOffset 
            },
            dateResult,
            timeResult,
            offset: offsetResult));
    }
}
